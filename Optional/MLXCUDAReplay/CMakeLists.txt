cmake_minimum_required(VERSION 3.25)
project(midnight_mlx_cuda_replay LANGUAGES C CXX)

option(MIDNIGHT_ENABLE_MLX_CUDA_REPLAY "Build the isolated experimental MLX replay probe" OFF)
if(NOT MIDNIGHT_ENABLE_MLX_CUDA_REPLAY)
  message(FATAL_ERROR "Replay is experimental and off by default. Explicitly enable MIDNIGHT_ENABLE_MLX_CUDA_REPLAY in an isolated build.")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The MLX CUDA integration probe requires Linux and CUDA.")
endif()
if(NOT EXISTS "${MLX_SOURCE}/mlx/backend/cuda/midnight_replay_impl.h")
  message(FATAL_ERROR "MLX_SOURCE must be an isolated source tree with the generated overlay applied.")
endif()

set(MLX_BUILD_CUDA ON CACHE BOOL "" FORCE)
set(MLX_BUILD_METAL OFF CACHE BOOL "" FORCE)
set(MLX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MLX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MLX_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(MLX_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MLX_SOURCE}" mlx)
target_compile_definitions(mlx PRIVATE MIDNIGHT_MLX_CUDA_REPLAY=1)
find_package(CUDAToolkit REQUIRED)
add_executable(midnight-mlx-replay-probe probe.cpp)
target_compile_features(midnight-mlx-replay-probe PRIVATE cxx_std_20)
target_compile_definitions(midnight-mlx-replay-probe PRIVATE MIDNIGHT_MLX_CUDA_REPLAY=1)
target_include_directories(midnight-mlx-replay-probe PRIVATE "${MLX_SOURCE}" "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(midnight-mlx-replay-probe PRIVATE mlx fmt::fmt-header-only CUDA::cudart CUDA::cuda_driver)

add_executable(midnight-mlx-replay-benchmark benchmark.cpp)
target_compile_features(midnight-mlx-replay-benchmark PRIVATE cxx_std_20)
target_compile_definitions(midnight-mlx-replay-benchmark PRIVATE MIDNIGHT_MLX_CUDA_REPLAY=1)
target_include_directories(midnight-mlx-replay-benchmark PRIVATE "${MLX_SOURCE}" "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(midnight-mlx-replay-benchmark PRIVATE mlx fmt::fmt-header-only CUDA::cudart CUDA::cuda_driver)
