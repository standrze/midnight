cmake_minimum_required(VERSION 3.25)
project(midnight_cuda_expert_probe LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The CUDA expert probe requires Linux and CUDA.")
endif()
if(NOT EXISTS "${MLX_SOURCE}/mlx/ops.h" OR NOT EXISTS "${MLX_LIBRARY}")
  message(FATAL_ERROR "Set MLX_SOURCE and MLX_LIBRARY to the isolated patched MLX core source and shared library.")
endif()
find_package(CUDAToolkit REQUIRED)
add_library(midnight_probe_mlx SHARED IMPORTED)
set_target_properties(midnight_probe_mlx PROPERTIES IMPORTED_LOCATION "${MLX_LIBRARY}")
add_executable(midnight-cuda-expert-probe probe.cpp)
target_compile_features(midnight-cuda-expert-probe PRIVATE cxx_std_20)
option(MIDNIGHT_MLX_CUDA_REPLAY "Match a combined core build that enables the replay overlay" OFF)
if(MIDNIGHT_MLX_CUDA_REPLAY)
  target_compile_definitions(midnight-cuda-expert-probe PRIVATE MIDNIGHT_MLX_CUDA_REPLAY=1)
endif()
target_include_directories(midnight-cuda-expert-probe PRIVATE "${MLX_SOURCE}" "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(midnight-cuda-expert-probe PRIVATE midnight_probe_mlx CUDA::cudart CUDA::cuda_driver)
